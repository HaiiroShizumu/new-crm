## 5. Target Architecture Specification

Задача: собственная CRM. Каждый выбор ниже закрывает конкретный вывод §1–§4.

### 5.1 Хранилище (PostgreSQL 16; то же переносится на MySQL 8)

```sql
-- CORE ---------------------------------------------------------------
CREATE TABLE app_user (
  id           bigserial PRIMARY KEY,
  email        citext NOT NULL UNIQUE,
  full_name    text   NOT NULL,
  is_active    boolean NOT NULL DEFAULT true,
  created_at   timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE org_unit (            -- закрывает «структуру компании» (§1.1: b_user_group)
  id bigserial PRIMARY KEY, parent_id bigint REFERENCES org_unit(id), name text NOT NULL
);
CREATE TABLE app_user_group (user_id bigint REFERENCES app_user(id), group_code text NOT NULL,
  PRIMARY KEY (user_id, group_code));   -- права: роль → операции, не «admin-флаг»

-- TASKS (закрывает §1.3) ---------------------------------------------
CREATE TABLE task (
  id            bigserial PRIMARY KEY,
  title         text NOT NULL,
  status        smallint NOT NULL,          -- 0 new .. 9 complete
  priority      smallint NOT NULL DEFAULT 2,
  responsible_id bigint NOT NULL REFERENCES app_user(id),
  created_by    bigint NOT NULL REFERENCES app_user(id),
  time_estimate_sec int,
  deadline      timestamptz,
  date_plan_start timestamptz, date_plan_end timestamptz,
  forum_topic_id bigint,                    -- или убрать: см. task_comment
  activity_at   timestamptz NOT NULL DEFAULT now(),   -- денорм. только ОДИН, пишется outbox-событием
  created_at    timestamptz NOT NULL DEFAULT now(),
  updated_at    timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ix_task_list ON task (deadline, status, responsible_id);
CREATE INDEX ix_task_activity ON task (activity_at DESC);

-- КОММЕНТАРИИ (закрывает §1.3: отдельная таблица, не чужой модуль)
CREATE TABLE task_comment (
  id        bigserial PRIMARY KEY,
  task_id   bigint NOT NULL REFERENCES task(id) ON DELETE CASCADE,
  parent_id bigint REFERENCES task_comment(id),
  author_id bigint NOT NULL REFERENCES app_user(id),
  body      text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  edited_at  timestamptz
);
CREATE INDEX ix_comment_task ON task_comment (task_id, created_at);
-- прочитанность: отдельная таблица, PK-подход Bitrix (§2, п. 6), НЕ счётчик в task
CREATE TABLE task_comment_seen (task_id bigint, user_id bigint, last_seen_at timestamptz NOT NULL,
  PRIMARY KEY (task_id, user_id));

-- ВРЕМЯ ЗАДАЧИ (закрывает §1.3: append-only + один таймер)
CREATE TABLE time_entry (                 -- журнал факта, агрегат = SUM / материализованное view
  id bigserial PRIMARY KEY,
  task_id bigint NOT NULL REFERENCES task(id) ON DELETE CASCADE,
  user_id bigint NOT NULL REFERENCES app_user(id),
  started_at timestamptz NOT NULL,
  ended_at   timestamptz,
  seconds    int NOT NULL CHECK (seconds >= 0),
  source     text NOT NULL,               -- 'timer' | 'manual' | 'day_close'
  note       text,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ix_time_entry_task ON time_entry (task_id, created_at);
CREATE INDEX ix_time_entry_user_day ON time_entry (user_id, (started_at::date));
CREATE TABLE time_timer (                 -- как b_tasks_timer: PK = пользователь (§2, п. 7)
  user_id    bigint PRIMARY KEY REFERENCES app_user(id),
  task_id    bigint NOT NULL REFERENCES task(id),
  started_at timestamptz NOT NULL,
  accumulator_sec int NOT NULL DEFAULT 0,
  CHECK (accumulator_sec >= 0)
);

-- РАБОЧИЙ ДЕНЬ (закрывает §1.4: активная запись гарантируется СХЕМОЙ, а не if-ом — §3, п. 7)
CREATE TABLE work_session (
  id bigserial PRIMARY KEY,
  user_id bigint NOT NULL REFERENCES app_user(id),
  status  text NOT NULL CHECK (status IN ('OPENED','PAUSED','CLOSED')),
  started_at timestamptz NOT NULL,
  paused_sec int NOT NULL DEFAULT 0,
  ended_at timestamptz,
  schedule_id bigint REFERENCES work_schedule(id),
  approved_at timestamptz,
  CHECK (status <> 'CLOSED' OR ended_at IS NOT NULL)
);
CREATE UNIQUE INDEX ux_work_session_one_active       -- вместо отсутствующего UNIQUE у Bitrix (§3, п. 7)
  ON work_session (user_id) WHERE status IN ('OPENED','PAUSED');
CREATE UNIQUE INDEX ux_work_session_no_overlap        -- анти-перекрытие
  ON work_session (user_id, daterange(started_at, ended_at, '[]'));

CREATE TABLE work_session_event (        -- журнал с REASON, как b_timeman_work_time_event_log (§2, п. 3)
  id bigserial PRIMARY KEY, session_id bigint NOT NULL REFERENCES work_session(id) ON DELETE CASCADE,
  event_type text NOT NULL,              -- START/PAUSE/CONTINUE/STOP/EDIT
  at timestamptz NOT NULL DEFAULT now(), actor_id bigint REFERENCES app_user(id), reason text
);

CREATE TABLE work_schedule (
  id bigserial PRIMARY KEY, name text NOT NULL,
  work_start time NOT NULL DEFAULT '09:00', work_end time NOT NULL DEFAULT '19:00',
  break_min int NOT NULL DEFAULT 60, work_days char(7) NOT NULL DEFAULT '1111100',
  auto_start boolean NOT NULL DEFAULT false, auto_close boolean NOT NULL DEFAULT false
);
CREATE TABLE work_shift_plan (
  id bigserial PRIMARY KEY, schedule_id bigint NOT NULL REFERENCES work_schedule(id),
  user_id bigint NOT NULL REFERENCES app_user(id), day date NOT NULL,
  shift_start timestamptz, shift_end timestamptz,
  UNIQUE (schedule_id, user_id, day)     -- как UNIQUE SHIFT_ID_USER_ID_DATE_ASSIGNED
);

CREATE TABLE user_day_plan (             -- вместо CUserOptions (§1.4: план дня в опции — хрупко)
  user_id bigint NOT NULL REFERENCES app_user(id), task_id bigint REFERENCES task(id),
  day date NOT NULL, position int NOT NULL DEFAULT 0,
  PRIMARY KEY (user_id, day, task_id)
);

-- PROCESS ENGINE (закрывает §1.2: состояние процесса, версии, журнал) --
CREATE TABLE wf_template (
  id bigserial PRIMARY KEY, code text NOT NULL UNIQUE,
  version int NOT NULL DEFAULT 1,
  dsl jsonb NOT NULL,                     -- узлы/рёбра/активности; вместо serialize+gzip BLOB
  status text NOT NULL DEFAULT 'published',   -- draft|published|pilot
  UNIQUE (code, version)
);
CREATE TABLE wf_instance (
  id bigserial PRIMARY KEY,
  template_id bigint NOT NULL REFERENCES wf_template(id),
  owner_type text NOT NULL, owner_id bigint NOT NULL,   -- документ-владелец (deal/task/...)
  state jsonb NOT NULL,                   -- checkpoint: позиция + переменные
  status text NOT NULL,                   -- RUNNING|SUSPENDED|COMPLETED|TERMINATED
  lock_owner text, lock_until timestamptz,             -- OWNER_ID/OWNED_UNTIL (§2, п. 1)
  started_by bigint, created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (owner_type, owner_id, template_id, status) WHERE status IN ('RUNNING','SUSPENDED')
);
CREATE INDEX ix_wf_instance_owner ON wf_instance (owner_type, owner_id);
CREATE TABLE wf_step_log (               -- вместо b_bp_tracking + b_tasks_log (§2, п. 3)
  id bigserial PRIMARY KEY, instance_id bigint NOT NULL REFERENCES wf_instance(id) ON DELETE CASCADE,
  step_code text NOT NULL, status text NOT NULL, result jsonb, error text,
  at timestamptz NOT NULL DEFAULT now()
) PARTITION BY RANGE (at);               -- партиционирование журнала по месяцам
CREATE TABLE wf_wait (                   -- вместо b_bp_scheduler_event + агентов (§1.2)
  id bigserial PRIMARY KEY, instance_id bigint NOT NULL REFERENCES wf_instance(id) ON DELETE CASCADE,
  event_code text NOT NULL, payload_filter jsonb, fire_at timestamptz,
  UNIQUE (instance_id, event_code, coalesce(payload_filter->>'k',''))
);

-- OUTBOX (закрывает §3, п. 1: событие НИКОГДА не обрабатывается в веб-запросе)
CREATE TABLE outbox (
  id bigserial PRIMARY KEY,
  topic text NOT NULL,                    -- task.updated | comment.added | wf.started ...
  payload jsonb NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  available_at timestamptz NOT NULL DEFAULT now(),
  attempts int NOT NULL DEFAULT 0, status text NOT NULL DEFAULT 'new'
);
CREATE INDEX ix_outbox_ready ON outbox (status, available_at) WHERE status IN ('new','retry');
-- единственная транзакция бизнес-записи ОБЯЗАТЕЛЬНО включает INSERT в outbox
```

Явно **не** копируем у Bitrix: 0 FK (здесь FK везде, каскады в БД), `serialize` BLOB (здесь `jsonb`),
`eval` (здесь интерпретатор DSL без исполнения произвольного PHP), 3 таблицы связки документ↔процесс
(здесь `wf_instance.owner_type/owner_id`), `COUNT`-на-лету для прочитанности без индекса
(здесь `task_comment_seen` + оконная функция), синхронный автозапуск процесса (здесь только outbox → воркер).

### 5.2 Границы сервисов (hexagonal)

```
interfaces/                     api/ (REST, admin UI) | cli/ | consumers/
application/                    сценарии = транзакции «команда → outbox»
   TaskService        (create/update/comment/complete)  -- как UpdateService: validate→save→history→outbox
   TimeTrackingService(startTimer/stopTimer/addManual)  -- инвариант «≤1 таймер на юзера» = PK
   WorkdayService     (start/pause/continue/close)      -- partial unique = «одна активная сессия»
   ProcessService     (start/subscribe/resume/signal)   -- только постановка в очередь
domain/                        сущности + чистые правила (статусы, SLA-расчёт рабочего времени)
infrastructure/                repositories (SQL), queue publisher, notifier, bitrix-adapter
workers/                       wf-stepper, outbox-dispatcher, session-autoclose, report-rollup
```

Правило: **никакой доменный код не вызывается из HTTP-контроллера напрямую** — контроллер только
валидирует и зовёт application-сервис, который пишет данные + outbox в одной транзакции.

### 5.3 Воркеры и очереди

- Транспорт: PostgreSQL `outbox` как source of truth + Redis Streams (или NATS) как доставка;
  воркеры: `dispatch` (топик → подписчики), `wf-stepper` (шаг процесса за прогон, лимит 1 документа —
  как `Stepper`), `session-autoclose` (по `fire_at`, аналог агента автозакрытия), `rollup` (ночные
  агрегаты часов/эффективности).
- Идемпотентность: `event_id` в payload, обработчик обязан быть `INSERT … ON CONFLICT DO NOTHING`
  по ключу. Порядок: партиция на топик = `aggregate_id % N`.
- Retry: backoff 1/5/10 мин (заимствуем расписание bizproc, `schedulerservice.php:665-673`),
  после 10 неудач — dead-letter таблица с ручным разбором.
- Расписание: `available_at` + `SELECT … FOR UPDATE SKIP LOCKED` (нет `eval`-агентов, нет крона «на каждый
  запрос», как у `Application::terminate`, §1.1).
- Backfill: пересчёт `activity_at`/счётчиков — отдельным job с курсором, не в веб-запросе (закрывает §3, п. 1).

### 5.4 API-контракты (REST + realtime)

```
POST /api/v1/tasks                {title, responsibleId, deadline, timeEstimateSec, dayPlan?:{day}}
PATCH /api/v1/tasks/{id}          If-Match: <updated_at>          → 409 при гонке
POST /api/v1/tasks/{id}/comments  {body, parentId?, mentions:[userId]}   → outbox: comment.added
POST /api/v1/tasks/{id}/time      {action:"start"|"stop"|"manual", seconds?, note?}
GET  /api/v1/tasks?status&responsible&deadlineFrom&cursor   → курсорная пагинация, без OFFSET
POST /api/v1/workday/action       {action:"start"|"pause"|"continue"|"close", at?, reason?}
GET  /api/v1/reports/worktime?from&to&userIds   → из work_session + work_session_event (не из «на лету»)
POST /api/v1/processes            {templateCode, owner:{type,id}, input}   → 202 Accepted,instanceId
POST /api/v1/processes/{id}/signal{event, payload}                        → очередь, 202
WS   /events   invalidate-события: {topic:"task.updated", id, version}    -- клиенты шарят список
```
Аутентификация — токен со scope (как REST 3.0: метод не в scope → `insufficient_scope`, метода нет → 404;
отсутствие сущности и «нет доступа» — одинаковый 404, как `note`-контроллеры, §4.4).

### 5.5 Маппинг «колёса Битрикс → готовое решение»

| Механизм Bitrix (вывод) | Частый in-house код | Что взять готовое |
|---|---|---|
| CBPWorkflow-интерпретатор + BLOB (§1.2) | своя «машина состояний» на if | `symfony/workflow` (state machine) или отдельный worker + JSON BPMN; для долгих human-процессов — Temporal/Cadence |
| b_agent с eval (§1.1) | крон-скрипты | `cron` + очередь (`golang-cron`, `scheduler` Horizon/Laravel, `APScheduler`) — без eval |
| EventManager синхронно (§1.1) | хуки в запросе | dispatcher + outbox (Debezium/CDC, если нужен реалтайм из чужих БД) |
| b_bp_script_queue + Stepper (§1.2) | «очередь в той же таблице сущности» | отдельная очередь: Redis Streams / SQS / RabbitMQ + DLQ |
| b_tasks_timer PK(USER_ID) (§1.3) | «флаг в строке пользователя» | partial unique index (§5.1) — оставить, это лучшее решение Битрикс |
| CUserOptions «план дня» (§1.4) | json в профиле | таблица `user_day_plan` (§5.1) |
| b_forum_message для комментариев (§1.3) | «переиспользуем чужой модуль» | своя таблица + готовый `flarum/laravel-<x>`-подобный слой или простая `task_comment` |
| Сериализация шаблонов serialize+gzip (§1.2) | сжатый PHP-массив | JSON Schema/BPMN + миграции версий (`wf_template.version`) |
| b_cache_tag / managed cache (§1.1) | свой кэш в репозиториях | Redis + invalidation по тегам (`cache-aside` + `DEL topic:{id}`), готово в стандартных фреймворках |
| REST v1/v3, scope-модель (§1.1, §4.4) | самописный API-слой | готовый API-гейтвей: FastAPI/Actix + scope-middleware, OpenAPI-генерация |
